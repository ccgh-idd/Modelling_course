1. Model Overview

We will use a simple SVIR compartmental model with these states:

  • S (Susceptible): Unvaccinated individuals at risk of infection
  • V (Vaccinated): Vaccinated individuals with reduced susceptibility
  • I (Infected): Currently infected individuals
  • R (Recovered): Individuals with immunity (either from vaccination or prior infection)

Simple SVIR Model

# ===== PARAMETERS (modify these to explore scenarios) =====
N <- 10000          # Total population
initial_infected <- 10
beta <- 0.5         # Transmission rate
gamma <- 1/5        # Recovery rate (1/infectious period in days)
vax_coverage <- 0.5 # Vaccination coverage (0-1)
vax_efficacy <- 0.8 # Vaccine efficacy against infection (0-1)

# Time parameters
days <- 365
dt <- 0.1
num_steps <- days / dt

# ===== INITIALIZE POPULATIONS =====
S <- N * (1 - vax_coverage)
V <- N * vax_coverage
I <- initial_infected
R <- N - S - V - I

# Storage for results
time_series <- numeric(num_steps)
s_history <- numeric(num_steps)
v_history <- numeric(num_steps)
i_history <- numeric(num_steps)
r_history <- numeric(num_steps)

# ===== RUN MODEL =====
for (step in 1:num_steps) {
  FOI <- beta * I / N
  
  S_to_I <- FOI * S * dt
  V_to_I <- FOI * (1 - vax_efficacy) * V * dt
  I_to_R <- gamma * I * dt
  
  S <- S - S_to_I
  V <- V - V_to_I
  I <- I + S_to_I + V_to_I - I_to_R
  R <- R + I_to_R
  
  time_series[step] <- step * dt
  s_history[step] <- S
  v_history[step] <- V
  i_history[step] <- I
  r_history[step] <- R
}

# ===== PLOT RESULTS =====
plot(time_series, s_history, type="l", lwd=2, col="blue", 
     xlab="Time (days)", ylab="Number of individuals",
     main="SVIR Model with Vaccination",
     ylim=c(0, max(s_history, v_history, i_history, r_history)))
lines(time_series, v_history, lwd=2, col="green")
lines(time_series, i_history, lwd=2, col="red")
lines(time_series, r_history, lwd=2, col="purple")
legend("right", legend=c("Susceptible", "Vaccinated", "Infected", "Recovered"),
       col=c("blue", "green", "red", "purple"), lwd=2)

# ===== RESULTS =====
total_infections <- R[num_steps] + I[num_steps]
peak_infections <- max(i_history)
attack_rate <- (total_infections / N) * 100

cat("\n=== RESULTS ===\n")
cat("Total infections:", round(total_infections, 0), "\n")
cat("Peak infections:", round(peak_infections, 0), "\n")
cat("Attack rate:", round(attack_rate, 2), "%\n")
